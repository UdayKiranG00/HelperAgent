import { google } from "@ai-sdk/google";
import { generateText, streamText, convertToModelMessages } from "ai";
import { z } from "zod";

const tools = {
  // 1. Backend executes automatically (Loops)
  runBackendTask: {
    description: "Run an automated backend routine.",
    parameters: z.object({}),
    execute: async () => ({ status: "Database query successful!" }),
  },
  // 2. Halts server-side loop (Triggers client UI)
  askHuman: {
    description: "Stop and ask the user for confirmation.",
    parameters: z.object({ question: z.string() }),
  },
};

export async function POST(req: Request) {
  const { messages } = await req.json();
  const currentMessages = await convertToModelMessages(messages);
  let needsHuman = false;

  // Sync Loop
  while (!needsHuman) {
    const { text, toolCalls } = await generateText({
      model: google("gemma-4-31b-it"),
      messages: currentMessages,
      tools,
    });

    currentMessages.push({
      role: "assistant",
      content: toolCalls.length > 0 
        ? [{ type: "text", text }, ...toolCalls.map(tc => ({ type: "tool-call", ...tc }))] as any
        : text,
    });

    if (toolCalls.length > 0) {
      const results = [];
      for (const call of toolCalls) {
        if (call.toolName === "askHuman") {
          needsHuman = true;
          break;
        }
        const execute = tools[call.toolName as keyof typeof tools]?.execute;
        if (execute) {
          results.push({
            type: "tool-result",
            toolCallId: call.toolCallId,
            toolName: call.toolName,
            result: await execute(),
          });
        }
      }
      if (needsHuman) break;
      currentMessages.push({ role: "tool", content: results as any });
    } else {
      needsHuman = true;
    }
  }

  // Stream output/UI trigger to client
  const result = streamText({
    model: google("gemma-4-31b-it"),
    messages: currentMessages,
    tools,
  });

  return result.toUIMessageStreamResponse();
}

"use client";

import { AssistantRuntimeProvider, makeAssistantToolUI } from "@assistant-ui/react";
import { useChatRuntime, AssistantChatTransport } from "@assistant-ui/react-ai-sdk";
import { Thread } from "@/components/thread"; // Adjust path to match your shadcn path

const AskHumanTool = makeAssistantToolUI({
  toolName: "askHuman",
  render: ({ args, status }) => {
    if (status.type === "complete") return <p className="text-zinc-500 italic">Confirmed</p>;
    
    return (
      <div className="p-4 border rounded bg-amber-50 dark:bg-zinc-900/50 my-2">
        <p className="font-semibold mb-2">{args.question}</p>
        <button 
          onClick={() => status.resolve({ approved: true })} 
          className="bg-black text-white dark:bg-white dark:text-black px-4 py-1.5 rounded text-xs font-bold"
        >
          Approve
        </button>
      </div>
    );
  },
});

export default function HomePage() {
  const runtime = useChatRuntime({
    transport: new AssistantChatTransport({ api: "/api/chat" }),
  });

  return (
    <AssistantRuntimeProvider runtime={runtime}>
      <div className="h-screen w-full">
        <Thread toolUIs={[AskHumanTool]} />
      </div>
    </AssistantRuntimeProvider>
  );
}
